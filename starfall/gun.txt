function gunFire()
    local ownerHead = owner():lookupBone("ValveBiped.Bip01_Head1")
    local bulletPos = owner():getBonePosition(ownerHead)
    local bullet = hologram.create(bulletPos,Angle(0,0,0),"models/hunter/blocks/cube025x025x025.mdl")
    local playerEyEangles = player():getEyeAngles()
    
    hook.add("Tick", "s", function()
        local v = Vector(500,0,0)
        v:rotate(playerEyEangles)
        bullet:setVel(v)
        bullet:setAngles(playerEyEangles)
    end)
end
