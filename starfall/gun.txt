local bullets = {}
local bulletSpeed = 8000
local bulletScale = Vector(1, 0.3, 0.3)

function gunFire()
    local bulletPos = owner():getEyePos() + owner():getVelocity() / 15
    local buletAng = owner():getEyeAngles()
    local bullet = hologram.create(bulletPos,Angle(),"models/hunter/blocks/cube025x025x025.mdl")
    local v = Vector(bulletSpeed,0,0)
           
    v:rotate(buletAng)
    bullet:setVel(v)
    bullet:setAngles(buletAng)
    bullet:setScale(bulletScale)   
    table.insert(bullets, #bullets+1, bullet)
end

function updateBullets(bullets)
    print("len"..#bullets)
    
    for k,v in pairs(bullets) do
         local traceFilte = {
            bullets[k],
            owner(), 
        }
        
        local dir,traceLen,startPos,endPos,t
        
        if bullets[k]:isValid() and bullets[k]:getPos():isInWorld() then
            dir = bullets[k]:getForward()
            traceLen = 5
            startPos = bullets[k]:getPos()
            endPos = startPos + dir * traceLen
            t = trace.line(startPos, endPos, traceFilte)
            
            if t.Hit then
                print(bullets[k])
                bullets[k]:remove()
                table.remove(bullets, k)
            end
        else
            print(bullets[k])
            bullets[k]:remove()
            table.remove(bullets, k)
        end
    end
end
